  \begin{frame}{Trusses vs frames}
    \textbf{Truss}
    \begin{itemize}
      \item Structure composed of oriented bar (rod) elements, connected by frictionless pins, carrying \textbf{axial forces} only.

    \begin{itemize}
      \item Structure composed of oriented beam elements, connected by welding,  carrying \textbf{transversal, axial forces and torsion}.

\section{Classic plate theory}

\begin{frame}{Plate structure}
  \begin{itemize}
    \item Plate structures are geometrically similar to structures of the 2D plane stress problem, but it usually carries only transversal loads that lead to \textbf{bending} deformation of the plate.
    \item For example: floors of a building, aerospace and ships structures, etc...
  \end{itemize}
  \begin{center}

\end{frame}

\begin{frame}{Plate models}
  \begin{columns}
    \begin{column}{0.48\textwidth}
      \textbf{Kirchhoff (1888) and Love (1945)}
      \begin{itemize}
        \item \emph{Shear free plates}: thin plates where the contribution of shear force on the deformations is neglected.
        \item Two-dimensional extension of the Bernoulli-Euler beam theory.
      \end{itemize}
    \end{column}

    \begin{column}{0.5\textwidth}
      \textbf{Mindlin (1951) and Reissner (1945)}
      \begin{itemize}
        \item \emph{Shear deformable plates}: thick plates where the contribution of shear force on deformations is considered.
        \item Two-dimensional extension of the Timoshenko beam theory.
      \end{itemize}
    \end{column}
  \end{columns}

\begin{frame}{Geometry assumptions}
  \begin{itemize}
    \item The thickness of the plate $h$ is constant and much smaller than the planar dimensions $a$ and $b$:
          $$h/a<0.1 \quad \text{and} \quad h/b<0.1.$$
    \item \textbf{Inextensibility of transverse fibers}: transverse normal strain is negligible.
          $$\varepsilon_{33} = 0.$$
  \end{itemize}
  \begin{center}
    \includegraphics[width=5cm]{../images/plateGeomAssumptions.jpeg}
  \end{center}
  \emph{\footnotesize (Credit:(G))}
\end{frame}

\begin{frame}{Material and loads assumptions}
  \textbf{Material}
  \begin{itemize}
    \item The material is homogenous and linear-elastic according to Hooke's law for a plane stress state $$\sigma_{33} = \sigma_{13} = \sigma_{23} = 0.$$
  \end{itemize}
  \textbf{Loads}
  \begin{itemize}
    \item External forces act only perpendicular to the $x-y$ plane, the vector of external moments lies within the $x-y$ plane.
    \item Displacement $u_3 (x, y,t)$ is small compared to $h$: $u_3 \lesssim 0.2h.$
  \end{itemize}
  \begin{center}
    \includegraphics[width=6cm]{../images/Plate.png}
  \end{center}
\end{frame}

\begin{frame}{Kirchhoff assumption}
  \textbf{Rectilinearity of the normals}: Bernoulli's hypothesis is valid, i.e. a cross-sectional plane stays plane and unwrapped in the deformed state.

  This means that the shear strains $\varepsilon_{13}$ and $\varepsilon_{23}$ due to the distributed shear forces $q_x$ and $q_y$ are neglected.
  \begin{center}
    \begin{center}
      \includegraphics[width=6cm]{../images/KirchhoffPlateHp.png}
    \end{center}
  \end{center}
  \begin{block}{}
    A straight fiber that is perpendicular to the middle plane of the plate before deformation remain straight and normal to it after deformation.\
  \end{block}
  \emph{\footnotesize (Credit: (N))}
\end{frame}

\begin{frame}{Strain-displacement relation}
  \begin{itemize}
    \item Using classical engineering definitions of strain: $$\varepsilon_{ii}=\partial_i u_i\qquad\text{and}\qquad\gamma_{ij}=\partial_i u_j+\partial_j u_i$$ we obtain
          $$\underbrace{\begin{bmatrix}
                \varepsilon_{11} \\ \varepsilon_{22} \\ \gamma_{12}
              \end{bmatrix}}_\text{$\boldsymbol{\varepsilon}$} = -z \underbrace{\begin{bmatrix}
                \partial^2_{xx} \\[7pt]
                \partial^2_{yy} \\[7pt]
                2\partial^2_{xy}
              \end{bmatrix}}_\text{$\boldsymbol{\nabla}_{k}$}u_3= z  \underbrace{\begin{bmatrix}
                \kappa_x \\ \kappa_y \\ \kappa_{xy}
              \end{bmatrix}}_\text{$\boldsymbol{\kappa}$}$$
    \item $\boldsymbol{\kappa}$ is the matrix that contains the changes in the \textbf{curvature of the plate}, given as $\boldsymbol{\kappa}=-\boldsymbol{\nabla}_{k}u_3$.
    \item Note that $\varepsilon_{12}=\varepsilon_{23}=0$ due to Kirchhoff assumptions and $\varepsilon_{33}=0$ due to the inextensibility of transverse fibers assumption.
  \end{itemize}
\end{frame}

\begin{frame}{Constitutive equation for isotropic material}
  \vspace{-0.3cm}
  \begin{itemize}
    \item  Classical plate theory assumes a plane stress state:  $$\sigma_{33} = \sigma_{13} =\sigma_{23} =0.$$
    \item  Constitutive equation for \emph{isotropic} material is $\boldsymbol{\sigma}=\mathbf C\boldsymbol{\varepsilon}$ or $\boldsymbol{\varepsilon}=\mathbf D\boldsymbol{\sigma}$ where
          \[
            \begin{bmatrix}
              \sigma_{11} \\
              \sigma_{22} \\
              \sigma_{12}
            \end{bmatrix}
            =
            \frac{E}{1 - \nu^2}
            \begin{bmatrix}
              1   & \nu & 0                 \\
              \nu & 1   & 0                 \\
              0   & 0   & \frac{1 - \nu}{2}
            \end{bmatrix}
            \begin{bmatrix}
              \varepsilon_{11} \\
              \varepsilon_{22} \\
              \gamma_{12}
            \end{bmatrix},
          \]
          or
          \[
            \begin{bmatrix}
              \varepsilon_{11} \\
              \varepsilon_{22} \\
              \gamma_{12}
            \end{bmatrix}
            =
            \frac{1}{E}
            \begin{bmatrix}
              1    & -\nu & 0          \\
              -\nu & 1    & 0          \\
              0    & 0    & 2(\nu + 1)
            \end{bmatrix}
            \begin{bmatrix}
              \sigma_{11} \\
              \sigma_{22} \\
              \sigma_{12}
            \end{bmatrix}.
          \]
    \item $\mathbf C$ is the \emph{elasticity matrix} and $\mathbf D=\mathbf C^{-1}$ is the \emph{elastic compliance matrix}.
  \end{itemize}
\end{frame}

\begin{frame}{Constitutive equation for orthotropic material}
  \begin{itemize}
    \item  The constitutive equation for \emph{orthotropic} material is $\boldsymbol{\sigma}=\mathbf C\boldsymbol{\varepsilon}$ or $\boldsymbol{\varepsilon}=\mathbf D\boldsymbol{\sigma}$ where
          \begin{equation*}
            \begin{bmatrix}
              \sigma_{11} \\
              \sigma_{22} \\
              \sigma_{12}
            \end{bmatrix}
            =
            \begin{bmatrix}
              Q_{11} & Q_{12} & 0      \\
              Q_{12} & Q_{22} & 0      \\
              0      & 0      & Q_{33}
            \end{bmatrix}
            \begin{bmatrix}
              \varepsilon_{11} \\
              \varepsilon_{22} \\
              \gamma_{12}
            \end{bmatrix}
          \end{equation*}
          where
          \begin{equation*}
            Q_{11} = \frac{E_1}{1 - \nu_{12} \nu_{21}}, \quad
            Q_{12} = \frac{\nu_{12} E_2}{1 - \nu_{12} \nu_{21}}, \quad
            Q_{22} = \frac{E_2}{1 - \nu_{12} \nu_{21}}, \quad
            Q_{33} = G_{12}
          \end{equation*}
    \item  The orthotropic properties of the lamina are given: $E_1$, $E_2$, $\nu_{12}$, $G_{12}$ and $\nu_{21} =\nu_{12}E_2/E_1$ applies.
  \end{itemize}
\end{frame}

  \vspace{0.3cm}
  \[\hspace{-0.3cm}
    \mathbf{M} =
    \begin{bmatrix}
      M_{11} \\
      M_{22} \\
      M_{12}
    \end{bmatrix}
    =
    \int_{-\frac{h}{2}}^{\frac{h}{2}} x_3
    \begin{bmatrix}
      \sigma_{11} \\
      \sigma_{22} \\
      \sigma_{12}
    \end{bmatrix}
    \, dx_3
    =
    \int_{-\frac{h}{2}}^{\frac{h}{2}} x_3 \, \boldsymbol{\sigma} \, dx_3
    =
    -\boldsymbol{C}\boldsymbol{\nabla}_ku_3\int_{-\frac{h}{2}}^{\frac{h}{2}} x_3^2  \, dx_3 =-\frac{h^3}{12}\boldsymbol{C}\boldsymbol{\nabla}_ku_3
  \]
\end{frame}


\begin{frame}{Dynamic equilibrium equation}
  \begin{itemize}
    \item Equilibrium condition for the vertical forces:
          $$\frac{\partial N_{13}}{\partial x_1}+\frac{\partial N_{23}}{\partial x_2}+f_3-\rho h \ddot{u}_3=0$$
    \item Equilibrium of moments:
          $$\frac{\partial M_{11}}{\partial x_1}+\frac{\partial M_{12}}{\partial x_2}-N_{13}=0$$
          $$\frac{\partial M_{22}}{\partial x_2}+\frac{\partial M_{12}}{\partial x_1}-N_{23}=0$$
  \end{itemize}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    Combining the three equations above, we obtain the \textbf{dynamic equilibrium equation for Kirchhoff-Love plate bending}:
    $$\boldsymbol{\nabla}_k^T \mathbf M+f_3=\rho h \ddot{u}_3$$
  \end{tcolorbox}
\end{frame}

\begin{frame}{Strong form for Kirchhoff-Love plate bending}
  Let $\Omega=[-a,a]\times[-b,b]$. Find the transverse displacement $u_{3}\in C^4(\Omega\times[0,T])$ such that
  \begin{equation*}
    \frac{h^3}{12}\boldsymbol{\nabla}_k^T \boldsymbol{C}\boldsymbol{\nabla}_ku_3+\rho h \ddot{u}_3=f_3 \qquad \text{  on  }\Omega\times]0,T[
  \end{equation*}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      Boundary conditions (\emph{simply supported}):
      \[
        \begin{aligned}
           & u_3 = 0\qquad \text{in }\partial\Omega\times]0,T[         \\
           & \mathbf M_n = 0\qquad \text{in }\partial\Omega\times]0,T[ \\
        \end{aligned}
      \]
    \end{column}

    \begin{column}{0.38\textwidth}
      Initial conditions:
      \[
        \begin{aligned}
           & \color{orange}{u_3(\cdot, 0) =  u_0 \quad \text{in }\Omega}      \\
           & \color{orange}{\dot{u}_3(\cdot, 0) = v_0 \quad \text{in }\Omega} \\
        \end{aligned}
      \]
    \end{column}
  \end{columns}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    In case of isotropic material, the strong form equation reduces to
    $$  D\left(
      \frac{\partial^4 u_3}{\partial x_1^4}
      + 2\frac{\partial^4 u_3}{\partial x_1^2 \partial x_2^2}
      + \frac{\partial^4 u_3}{\partial x_2^4}
      \right)   + \rho h \ddot{u}_3
      = f_3$$
    where $D=Eh^3/(12(1-\nu^2))$.
  \end{tcolorbox}
\end{frame}
}

\begin{frame}{Weak form for Kirchhoff-Love simply supported plate}
  The weak form consists of finding the transverse displacement $u_{3}\in \mathcal U$ such that the following equation is satisfied for every virtual transversal displacement $\delta u_3\in \mathcal V$:
  \begin{block}{}
    \[
      \frac{h^3}{12}\int_\Omega \, (\boldsymbol{\nabla}_k \delta u_3)^T\mathbf C \, \boldsymbol{\nabla}_k u_3  \, d\Omega + \int_\Omega \rho h\, \ddot u_3 \, \delta u_3 \, d\Omega = \int_\Omega f_3\, \delta u_3 \, d\Omega
    \]
  \end{block}

  The functional spaces are defined as
  \begin{align*}
    \mathcal U & =\big\{u_3(\cdot,t)\in H^2(\Omega)\ |\ u_3=0\text{ in } \partial\Omega\times]0,T[\big\} \\[0.1cm]
    \mathcal V & =\big\{\delta u_3\in H^2(\Omega)\ |\ \delta u_3=0\text{ in } \partial\Omega\big\}
  \end{align*}
\end{frame}

    \item ${}^e\mathbf K$ and ${}^e\mathbf M$ are ($12\times 12$) matrices and ${}^e\mathbf r$ is a ($12\times 1$) vector:
          \begin{align*}
            {}^e\mathbf K & =\frac{h^3}{12}\int_\Omega {}^a\mathbf B^T\mathbf C{}^a\mathbf B \, d\Omega=\frac{h^3ab}{12}\int_{-1}^1\int_{-1}^1 {}^a\mathbf B^T\mathbf C{}^a\mathbf B\, d\xi_1d\xi_2, \\
            {}^e\mathbf M & =\int_\Omega \rho h\, {}^a\mathbf H^T{}^a\mathbf H \, d\Omega=\rho hab \int_{-1}^1\int_{-1}^1 {}^a\mathbf H^T {}^a\mathbf H\, d\xi_1d\xi_2,                              \\
            \mathbf r(t)  & =\int_{\Omega} {}^a\mathbf H^{T}f_3(t)\, d\Omega=ab f_3(t)\int_{-1}^1\int_{-1}^1 {}^a\mathbf H^T \, d\xi_1d\xi_2.
          \end{align*}
  \end{itemize}
\end{frame}
}

\begin{frame}{Post processing}
  \begin{itemize}
    \item The \textbf{approximated stresses} ${}^e\sigma_{11}^h$ and ${}^e\sigma_{22}^h$ and ${}^e\sigma_{12}^h$ at any point $(x_1,x_2,x_3)$ of the element $e$ are given in term of the nodal displacements:
          \vspace{0.3cm}
          $${}^e\boldsymbol{\sigma}^h=\mathbf C{}^e\boldsymbol{\varepsilon}^h=-x_3\mathbf C\boldsymbol{\nabla}_k {}^eu_3^h=-x_3\mathbf C{}^e\mathbf B{}^e\mathbf{q}(t)$$
          \vspace{-0.3cm}
    \item The \textbf{approximated bending moments} ${}^eM_{11}^h$ and ${}^eM_{22}^h$ and twisting moment ${}^eM_{12}^h$ per unit length are given by:
          \vspace{0.3cm}
          \[{}^e\mathbf M=\begin{bmatrix}
              {}^eM_{11}^h \\
              {}^eM_{22}^h \\
              {}^eM_{12}^h
            \end{bmatrix}
            =
            \int_{-\frac{h}{2}}^{\frac{h}{2}} x_3
            \begin{bmatrix}
              {}^e\sigma_{11}^h \\
              {}^e\sigma_{22}^h \\
              {}^e\sigma_{12}^h
            \end{bmatrix}
            \, dx_3 =-\frac{h^3}{12}\boldsymbol{C}{}^e\mathbf B{}^e\mathbf{q}(t)
          \]
  \end{itemize}
\end{frame}

\begin{frame}{Comparison: AMC vs CR plate bending element}
  \textbf{AMC element (\emph{Non conforming})}
  \begin{itemize}
    \item 3 degrees of freedom per node: \( u_3\), \( \theta_1 \), \( \theta_2 \).
    \item Only displacement \( u_3 \) is continuous across elements; rotations may have jumps.
    \item Nonconforming element: does not fully satisfy \( C^1 \) continuity.
    \item Simpler and computationally cheaper; suitable for practical applications.
  \end{itemize}
  \vspace{0.5cm}

  \textbf{CR element (\emph{Conforming})}
  \begin{itemize}
    \item 4 degrees of freedom per node: \( u_3 \), \( \theta_1 \), \( \theta_2 \), and \(  \theta_{12} \).
    \item Displacement \( u_3 \) and rotations \( \theta_1, \theta_2 \) are continuous across element boundaries.
    \item Fully conforming to the \( C^1 \) continuity required by Kirchhoff plate theory.
    \item Higher computational cost and complexity.
  \end{itemize}
\end{frame}

\begin{frame}{\emph{Why would you use AMC element if it's nonconforming?}}
  \begin{itemize}
    \item Computationally cheaper than full $C^1$ elements (fewer DOFs).
    \item It still converges (theoretical results for nonconforming FEMs show convergence \emph{under certain conditions}).
    \item Suitable when small slope discontinuities are acceptable (e.g., dynamic problems, large meshes).
    \item Tends to underestimate the natural frequencies, making it a useful benchmark for detecting overstiffness in numerical model.
  \end{itemize}
\end{frame}
