\begin{frame}{Eliminating prescribed degrees of freedom}
  \small Group the nodal components into free unknowns $\mathbf q_f$ and prescribed
  values $\mathbf q_c=\hat{\mathbf q}_c(t)$ on $\Gamma_u$:
  \[
    \begin{bmatrix}\mathbf M_{ff}&\mathbf M_{fc}\\\mathbf M_{cf}&\mathbf M_{cc}\end{bmatrix}
    \begin{bmatrix} \ddot{\mathbf q}_f\\ \ddot{\mathbf q}_c\end{bmatrix}+\begin{bmatrix}\mathbf K_{ff}&\mathbf K_{fc}\\\mathbf K_{cf}&\mathbf K_{cc}\end{bmatrix}
    \begin{bmatrix}\mathbf q_f\\\mathbf q_c\end{bmatrix}
    =\begin{bmatrix}\mathbf r_f\\\mathbf r_c\end{bmatrix}.
  \]
  For the free rows we obtain the system
  \[
    \mathbf M_{ff}\ddot{\mathbf q}_f + \mathbf M_{fc}\ddot{\hat{\mathbf q}}_c
    + \mathbf K_{ff}\mathbf q_f + \mathbf K_{fc}\hat{\mathbf q}_c = \mathbf r_f .
  \]
  \vspace{-0.4cm}
  \begin{block}{Reduced equations of motion}
    \[
      \mathbf M_{ff}\ddot{\mathbf q}_f+\mathbf K_{ff}\mathbf q_f
      =\underbrace{\mathbf r_f-\mathbf M_{fc}\ddot{\hat{\mathbf q}}_c
        -\mathbf K_{fc}\hat{\mathbf q}_c}_{\mathbf r_{\mathrm{eff}}(t)} .
    \]
  \end{block}
  \begin{itemize}
    \item For fixed supports, $\hat{\mathbf q}_c=\mathbf 0$ and $\mathbf r_{\mathrm{eff}}=\mathbf r_f$.
    \item \(\ddot{\hat{\mathbf q}}_c(t)\) is the prescribed support acceleration and it can be computed from $\hat{\mathbf q}_c$.
    \item Integrate from $\mathbf q_f(0)=\mathbf u_{0,f}$ and $\dot{\mathbf q}_f(0)=\mathbf v_{0,f}$.
  \end{itemize}
\end{frame}

\begin{frame}{Harmonic response}
  \small To simplify the notation let us write $\mathbf M=\mathbf M_{ff}$, $\mathbf K=\mathbf K_{ff}$, $\mathbf q=\mathbf q_{f}$ and $\mathbf r=\mathbf r_{\mathrm{eff}}$.

  Let $n$ be the number of free degrees of freedom.
  \begin{block}{Free vibration solution}
    The solution for a \emph{free} ($\mathbf r= \mathbf 0$) and \emph{undamped} discrete vibration problem :
    $$\mathbf{M} \ddot{\mathbf{q}}(t) + \mathbf{K} \mathbf{q}(t) = \mathbf{0}$$
    is sinusoidal, described via an harmonic function:
    $$\mathbf q(t)=\alpha \mathbf p \cos(\omega t+\varphi)$$
    \begin{multicols}{2}
      \begin{itemize}
        \item \(\mathbf{p}\): mode shape (modal vector)
        \item \(\omega\): angular frequency
        \item \(\varphi\): phase
        \item \(\alpha\): scaling factor
      \end{itemize}
    \end{multicols}
  \end{block}
  \begin{itemize}
    \item  \textcolor{darkred}{\textbf{Solutions are defined up to a scalar factor: $\alpha$.}}
    \item Three scalars unknowns $\alpha$, $\omega$, $\varphi$  and one unknown vector $\mathbf{p}$.
  \end{itemize}
\end{frame}

\begin{frame}{Generalized eigenvalue problem}
  \small
  Substituting the proposed ansatz into the reduced equation of motion yields a generalized eigenvalue problem:
  \vspace{-0.2cm}
  \begin{align*}
    \alpha (\mathbf{K} - \omega^2 \mathbf{M}) \mathbf{p} \cos(\omega t - \varphi) & = 0 \\
    (\mathbf{K} - \omega^2 \mathbf{M}) \mathbf{p}                                 & = 0
  \end{align*}
  \vspace{-0.8cm}
  \begin{block}{}
    \textbf{Solving the generalized eigenvalue problem:}
    \begin{itemize}
      \item \emph{Eigenvalues}: $\lambda_j=\omega_j^2$ for $j=1,\dots,n$ are the roots of the characteristic polynomial:
            $$\text{det}(\mathbf K-\omega^2\mathbf M)=0.$$
      \item \emph{Eigenvectors}: $\mathbf p_j$ for $j=1,\dots,n$ are the solution of the equation
            $$(\mathbf{K} - \lambda_j \mathbf{M})\mathbf p_j=\mathbf 0.$$
    \end{itemize}
  \end{block}
\end{frame}


\begin{frame}{General solution of free vibrations via modal superposition}
  \small
  \begin{itemize}
    \item Once the sets of eigenvalue-eigenvector pairs \((\omega_j^2, \mathbf{p}_j)\) for $j=1,\dots,n$ have been determined, the linearity of the system allows the \textbf{general solution for free vibrations} to be expressed as a superposition of modal contributions:
          \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
            \[
              \mathbf{q}(t) = \sum_{j=1}^n \alpha_j \, \mathbf{p}_j \cos(\omega_j t + \varphi_j)
            \]
          \end{tcolorbox}
    \item The constants \(\alpha_j\) and \(\varphi_j\) are determined from the initial conditions: $\mathbf{q}(0) = \mathbf{u}_0$ and $\dot{\mathbf{q}}(0) = \mathbf{v}_0$ leading to the system:
          \begin{align*}
            \sum_{j=1}^n \alpha_j \mathbf{p}_j \cos(\varphi_j)          & = \mathbf{u}_0  \\
            \sum_{j=1}^n \alpha_j \omega_j \mathbf{p}_j \sin(\varphi_j) & = -\mathbf{v}_0
          \end{align*}
  \end{itemize}
\end{frame}

\begin{frame}{Exact modal response to a time-dependent load}
  \small Consider the reduced equations of motion with a non-zero forcing term:
  \[
    \mathbf M\ddot{\mathbf q}+\mathbf K\mathbf q
    =\mathbf r(t),
  \]
  subject to zero initial displacement and velocity conditions : $\mathbf q(t=0)=\dot{\mathbf q}(t=0)=\mathbf{0}$.
  \begin{enumerate}
    \item Compute the eigenvectors $\mathbf p_j$ and eigenvalues $\lambda_j$ of the associated eigenvalue problem.
    \item  Change coordinates to $\mathbf q=\mathbf P\mathbf z$, where $\mathbf z$ contains the \emph{modal coordinates} and
          $$\mathbf P=\left[ \begin{array}{c;{2pt/2pt}c;{2pt/2pt}c;{2pt/2pt}c}
                \mathbf p_{1} & \mathbf p_{2} & \dots & \mathbf p_{n} \\
              \end{array}\right]$$
          is called the \emph{modal matrix}.
    \item Premultiplying the reduced equations of motion by $\mathbf P^T$ gives
          $$\underbrace{\mathbf P^T\mathbf M\mathbf P}_{\mathbf I}\,\ddot{\mathbf z}+\underbrace{\mathbf P^T\mathbf K\mathbf P}_{\mathbf\Lambda}\,\mathbf z=\mathbf P^T\mathbf r(t).$$
          where $\mathbf I$ is the identity matrix and $\mathbf\Lambda$ the \emph{spectral matrix}:
          $$\mathbf\Lambda=\text{diag}(\lambda_1,\dots,\lambda_n)=\text{diag}(\omega_1^2,\dots,\omega_n^2).$$
  \end{enumerate}
\end{frame}

\begin{frame}{Exact modal response to a time-dependent load}
  \small \begin{enumerate}
    \setcounter{enumi}{3}
    \item The equations of motion therefore decouple into independent modal equations, which can be solved exactly:
          \[
            \ddot z_j+\omega_j^2 z_j=\mathbf p_j^T\mathbf r(t).
          \]
          The exact unconstrained displacement vector is
  \end{enumerate}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    $$\mathbf q(t)=\sum_{j=1}^{n}\mathbf p_j\underbrace{ \frac{1}{\omega_j}\int_0^t \mathbf p^T_j\mathbf r_{j}(\tau)\sin(\omega_j(t-\tau))\, d\tau }_{z_j(t)}$$
  \end{tcolorbox}
  \small \begin{enumerate}
    \setcounter{enumi}{4}
    \item Exact modal response to non-zero initial conditions at $t=0$:
          \[
            \mathbf q(t)
            =\sum_{j=1}^{n}\mathbf p_j\left[
              z_j(0)\cos(\omega_j t)
              +\dot z_j(0)\frac{\sin(\omega_j t)}{\omega_j}
              +\frac{1}{\omega_j}\int_0^t
              \mathbf p_j^T\mathbf r_(\tau)
              \sin\bigl(\omega_j(t-\tau)\bigr)\,d\tau
              \right].
          \]
          The initial modal displacement and velocity are
          $z_j(0)=\mathbf p_j^T\mathbf M\mathbf u_{0}$ and
          $\dot z_j(0)=\mathbf p_j^T\mathbf M\mathbf v_{0}$.
  \end{enumerate}

\end{frame}


\begin{frame}{Why low-frequency modes often dominate displacement}
  \vspace{-0.2cm}
  \small \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    $$\mathbf q(t)=\sum_{j=1}^{n}\mathbf p_jz_j(t)=\sum_{j=1}^{n}\mathbf p_j\frac{1}{\omega_j}\int_0^t \mathbf p^T_j\mathbf r_{j}(\tau)\sin(\omega_j(t-\tau))\, d\tau$$
  \end{tcolorbox}
  For a harmonic load $\mathbf r(t)=\hat{\mathbf r}\cos(\Omega t)$, where $\hat{\mathbf r}$ is a constant, the particular solution is
  \[
    z_j(t)=
    \frac{\mathbf p_j^T\hat{\mathbf r}}{\omega_j^2-\Omega^2}\cos(\Omega t).
  \]
  \vspace{-0.4cm}
  \begin{itemize}
    \item \textbf{Slow load.} When $\Omega\ll\omega_j$, the modal amplitude
          scales approximately as $1/\omega_j^2$ for equal modal forces. Doubling
          the natural frequency therefore reduces the amplitude by about a factor
          of four.
    \item \textbf{Load location and direction.} If
          $\mathbf p_j^T\hat{\mathbf r}=0$, the load has no component in this mode
          and therefore cannot excite it, however low its frequency.
    \item \textbf{Resonance.} When $\Omega$ approaches $\omega_j$, that mode can
          dominate the response. An abrupt load can excite several modes and is not
          restricted to the low-frequency ones.
  \end{itemize}
\end{frame}

\section{Fundamental properties of generalized eigenvalue problems}

\begin{frame}{Spectral properties}
  \small
  \begin{itemize}
    \item If both $\mathbf{K}$ and $\mathbf{M}$ are symmetric and strictly positive definite, then all eigenvalues $\lambda_j$ of the generalized eigenvalue problem
          $$\mathbf{K} \mathbf{p}_j = \lambda_j \mathbf{M} \mathbf{p}_j$$
          are \textbf{real and positive}.
    \item We define the angular frequencies:
          $$\omega_j=\sqrt{\lambda_j}\quad [rad/s]$$
          and
          $$0<\omega_1\leq \omega_2\leq\omega_3\leq\dots\leq\omega_n.$$
    \item To obtain the frequencies in hertz, divide the angular frequencies by $2\pi$:
          $$f_j=\frac{\omega_j}{2\pi}\quad [Hz].$$
  \end{itemize}
\end{frame}

\begin{frame}{Rigid body modes}
  \small In the semi-discrete weak form obtained via finite element discretization:
  \begin{itemize}
    \item The \textbf{mass matrix} \( \mathbf{M} \) is symmetric and strictly positive definite.
    \item The \textbf{stiffness matrix} \( \mathbf{K} \) is symmetric and positive semi-definite:
          \[
            \mathbf{K} \mathbf{p} = \mathbf{0} \quad \text{for certain nonzero vectors } \mathbf{p}.
          \]
  \end{itemize}
  Consequently, the eigenvalues \( \omega_j^2 \) of the generalized eigenvalue problem are all real and non-negative:
  \[
    0 \,\leq\, \omega_1 \leq \omega_2 \leq \dots \leq \omega_n.
  \]

  \vspace{0.2cm}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    \textbf{Rigid body modes:} zero eigenvalues (i.e., \( \omega_j = 0 \)) correspond to \emph{rigid body motions}, where the system undergoes displacement without internal deformation.
  \end{tcolorbox}
\end{frame}
}

\begin{frame}{Spectral offset technique}
  \vspace{-0.2cm}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    \small To ensure that the stiffness matrix is \emph{strictly} positive definite, a \textbf{spectral shift} is employed. Given \( \sigma > 0 \), define the modified stiffness matrix as:
    \[
      \mathbf{K}_\sigma = \mathbf{K} + \sigma \mathbf{M}
    \]
    $${\textit{Empirical rule:}} \qquad \sigma\approx \frac{1}{100}\frac{\text{tr}(\mathbf K)}{\text{tr}(\mathbf M)}$$
  \end{tcolorbox}
  \begin{itemize}
    \small
    \item Consider the offset eigenvalue problem:
          $            (\mathbf{K}_\sigma - \lambda_\sigma \mathbf{M}) \mathbf{p} = \mathbf{0}
          $
    \item Substituting the definition of \( \mathbf{K}_\sigma \):
          \[
            (\mathbf{K} + \sigma \mathbf{M} - \lambda_\sigma \mathbf{M}) \mathbf{p}
            = (\mathbf{K} + (\sigma - \lambda_\sigma)\mathbf{M}) \mathbf{p} = \mathbf{0}
          \]
    \item Comparing with \( (\mathbf{K} -\lambda \mathbf{M}) \mathbf{p}=\mathbf 0 \), we identify: $\lambda_\sigma = \lambda+\sigma > 0$.
  \end{itemize}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    \centering \textbf{Eigenvalues are shifted by \( \sigma \), the eigenvectors remain the same.}
  \end{tcolorbox}

\end{frame}

\begin{frame}{Orthogonality of mode shapes}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    Let $\mathbf p_i$ and $\mathbf p_j$ two eigenvectors corresponding to two \emph{distinct} eigenvalues $\lambda_i$ and $\lambda_j$, then
    $$
      \mathbf p_i^T\mathbf M\mathbf p_j  =0
      \qquad\text{and}\qquad
      \mathbf p_i^T\mathbf K\mathbf p_j  =0\qquad\ (i\neq j).
    $$
  \end{tcolorbox}
  \emph{Consequence:} two different harmonic responses:
  $$\mathbf q_i(t)=\alpha_i \mathbf p_i \cos(\omega_i t+\varphi_i)\qquad \text{and}\qquad \mathbf q_j(t)=\alpha_j \mathbf p_j \cos(\omega_j t+\varphi_j)$$
  are $\mathbf M$- and $\mathbf K$-orthogonal:
  $$\mathbf q_i^T\mathbf M\mathbf q_j=0\qquad\text{and}\qquad \mathbf q_i^T\mathbf K\mathbf q_j=0.$$
  \vspace{-0.5cm}
  \begin{block}{}
    \centering The virtual work of inertial and elastic forces of a given mode, along the displacement given by a different mode, is zero.
  \end{block}
\end{frame}

\begin{frame}{Normalization of mode shapes}

  Let $\mathbf p_i$ and $\mathbf p_j$ two eigenvectors corresponding to the eigenvalues $\lambda_i$ and $\lambda_j$, then
  $$
    \mathbf p_i^T\mathbf M\mathbf p_j  =\mu_i \delta_{ij}
    \qquad\text{and}\qquad
    \mathbf p_i^T\mathbf K\mathbf p_j  =\kappa_i\delta_{ij}
  $$
  where
  \begin{itemize}
    \item $\mu_i=$ modal mass of mode $i$,
    \item $\kappa_i=$ modal stiffness of mode $i$,
    \item $\delta_{ij}=$ Kronecker delta.
  \end{itemize}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    \textbf{Usual orthonormalization choice: }
    \[
      \mu_i = 1 \qquad  \Rightarrow \qquad \lambda_i = \frac{\kappa_i}{\mu_i} = \omega_i^2.
    \]
  \end{tcolorbox}
\end{frame}

\setbeamercolor{itemize/enumerate body}{fg=white}
\color{white}
\begin{frame}{Orthonormalization of mode shapes}
  \vspace{-0.2cm}
  \begin{block}{}
    Let $\mathbf p_i$ and $\mathbf p_j$ two eigenvectors corresponding to the  eigenvalues $\lambda_i$ and $\lambda_j$, then
    $$
      \mathbf p_i^T\mathbf M\mathbf p_j  =\delta_{ij}
      \qquad\text{and}\qquad
      \mathbf p_i^T\mathbf K\mathbf p_j  =\omega_i^2\delta_{ij}
    $$
    where $\delta_{ij}$ represent Kronecker symbol.
  \end{block}
  \vspace{-0.3cm}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    We store the modal vectors $\mathbf p_i$ in a so-called \emph{modal matrix} $\mathbf P$:
    $$\mathbf P=\left[ \begin{array}{c;{2pt/2pt}c;{2pt/2pt}c;{2pt/2pt}c}
          \mathbf p_{1} & \mathbf p_{2} & \dots & \mathbf p_{n} \\
        \end{array}\right]$$
    then
    $$  \mathbf P^T\mathbf M\mathbf P  =\mathbf I       \qquad\text{and}\qquad
      \mathbf P^T\mathbf K\mathbf P  =\mathbf\Lambda$$
    where $\mathbf I$ is the identity matrix of order $n$ and $\mathbf\Lambda$ the spectral matrix:
    $$\mathbf\Lambda=\text{diag}(\lambda_1,\dots,\lambda_n)=\text{diag}(\omega_1^2,\dots,\omega_n^2).$$
  \end{tcolorbox}
\end{frame}
}

\begin{frame}{Transformation to standard form by decomposition}
  To reduce to a standard eigenvalue problem:
  \vspace{0.2cm}
  \begin{itemize}
    \item Let \( \mathbf{L} \) be the \emph{Cholesky factor} of \( \mathbf{M} \): \( \mathbf{M} = \mathbf{L} \mathbf{L}^T\).
          \begin{block}{}
            Since $\mathbf M$ is real, symmetric and positive-definite, $\mathbf L$ is a real lower triangular matrix with positive diagonal entries.
          \end{block}
    \item Define the change of variables \( \mathbf{v} = \mathbf{L}^T \mathbf{p} \). Substituting into the original problem and pre-multiplying by \( \mathbf{L}^{-1} \) and pirt-multiplying by \( \mathbf{L}^{-T} \) yields:
          \[
            (\mathbf{K} - \omega^2 \mathbf{L} \mathbf{L}^T) \mathbf{p} =0
            \quad \Rightarrow \quad
            (\mathbf L^{-1}\mathbf{K}\mathbf L^{-T} - \omega^2 \mathbf{I}) \mathbf{v}=0.
          \]
    \item  \textbf{Result:} standard eigenvalue problem for symmetric and positive definite matrix \( \mathbf A=\mathbf L^{-1}\mathbf{K}\mathbf L^{-T}\):
          \[
            (\mathbf A - \lambda \mathbf{I})\mathbf v=\mathbf 0.
          \]
  \end{itemize}
\end{frame}

\begin{frame}{Transformation to standard form by decomposition}
  \begin{itemize}
    \item   Cholesky decomposition can be applied to $\mathbf K$ as well, since $\mathbf K$ is symmetric positive-definite.
          \[
            \mathbf{K} = \mathbf{L} \mathbf{L}^T.
          \]
    \item  \textbf{Result:} standard eigenvalue problem for symmetric and positive definite matrix \( \mathbf A=\mathbf L^{-1}\mathbf{M}\mathbf L^{-T}\), eigenvalues $\theta=1/\lambda$ and eigenvectors $\mathbf{v} = \mathbf{L}^T \mathbf{p}$:
          \[
            (\mathbf A - \theta \mathbf{I})\mathbf v=\mathbf 0.
          \]
  \end{itemize}
  \begin{tcolorbox}[colback=darkred!10,colframe=darkred!40!black]
    For a matrix of size $n\times n$, the computational cost of Cholesky decomposition is $O(n^3)$.
  \end{tcolorbox}
\end{frame}

\begin{frame}{Small and large scale generalized eigenvalue problems}
  \small We are interested in economical computational algorithms for extracting eigenpairs $(\lambda_j,\mathbf p_j)$ of the generalized eigenvalue problem :
  $$(\mathbf K - \lambda \mathbf{M})\mathbf p=\mathbf 0$$
  \vspace{-0.8cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      \begin{block}{}
        \textbf{\centering Small-scale problems}
        \begin{itemize}
          \item Matrices are of modest size $n\leq 250$ or $250\leq n \leq 2500$ and small bandwidth matrices.
          \item An explicit reduction to standard eigenvalue form is typically employed.
          \item The most effective algorithms are:
                \begin{enumerate}
                  \item Rayleigh quotient minimization,
                  \item Generalized Jacobi method,
                  \item Householder-QR method.
                \end{enumerate}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{.48\textwidth}
      \begin{block}{}
        \textbf{Large-scale problems}
        \begin{itemize}
          \item $n\geq 2500$.
          \item The most effective algorithms are :
                \begin{enumerate}
                  \item Subspace iteration,
                  \item Lanczos' method,
                  \item Guyan-Irons method,
                  \item Arnoldi Davidson methods.
                \end{enumerate}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Large-scale eigenproblems reduction via Rayleigh--Ritz procedure}
  \small
  We seek the \textbf{only the lowest} $n_{\mathrm{modes}}\ll n$ eigenpairs of
  $$(\mathbf K - \lambda \mathbf{M})\mathbf p=\mathbf 0$$
  \vspace{-0.6cm}
  \begin{enumerate}
    \item  Choose $q$ independent static load patterns, with $n_{\mathrm{modes}}\le q\ll n$:
          \[
            \mathbf L=[\boldsymbol\ell_1,\ldots,\boldsymbol\ell_q]\in\mathbb R^{n\times q}.
          \]
    \item \textbf{Use static loads as trial vectors.} For nonsingular $\mathbf K$, factorize it once and solve for all $q$ load cases:
          \[
            \mathbf K\mathbf R=\mathbf L,\qquad
            \mathbf R=[\mathbf r_1,\ldots,\mathbf r_q]\in\mathbb R^{n\times q}.
          \]
  \end{enumerate}
  \begin{block}{}
    Guidelines for selecting $\mathbf L$ are vague. Select loads that excite the target motions. For example, transverse loads on a flexible beam span capture bending, while a torque can capture torsion. A mode orthogonal to every load pattern is absent from the static trial space in the modal expansion.
  \end{block}

\end{frame}

\begin{frame}{Large-scale eigenproblems: projection and recovery}
  \vspace{-0.2cm}
  \small
  \begin{enumerate}
    \setcounter{enumi}{3}
    \item   The displacements $\mathbf R$, referred as \emph{trial vectors}, are used to form the reduced eigenproblem of size $q$ :
          $$(\mathbf K^* - \lambda^* \mathbf{M}^*)\mathbf p^*=\mathbf 0$$
          The projected matrices $\mathbf K^*$ and $\mathbf M^*$ are generally dense, but small ($q\times q$).
          $$\mathbf K^*=\mathbf R^T\mathbf K\mathbf R\quad \mbox{and}\quad \mathbf M^*=\mathbf R^T\mathbf M\mathbf R$$
  \end{enumerate}
  \vspace{-0.2cm}
  \begin{block}{}
    \begin{enumerate}
      \setcounter{enumi}{4}
      \item Solve the small-scale generalize eigenvalue problem and find \emph{approximated projected eigenpairs} $(\lambda_j^*,\mathbf p_j^*)$ for $j=1\dots,q$. The $\lambda_j^*$ are referred to as \textbf{Ritz values}.
    \end{enumerate}
  \end{block}
  \begin{enumerate}
    \setcounter{enumi}{5} \item To recover the approximated mode shapes and angular frequencies:
          \[
            {\mathbf p}_j=\mathbf R\mathbf p_j^*,\qquad
            \lambda_j=\lambda_j^*.
          \]
    \item   Check if $\mathbf K{\mathbf p}_j-\lambda_j\mathbf M{\mathbf p}_j=\mathbf 0$ in the full system. Add trial vectors if the residuals exceed the chosen tolerance.
  \end{enumerate}
\end{frame}

\section[Numerical modal extraction algorithms for small-scale problems]{Numerical modal extraction algorithms}

\begin{frame}{Rayleigh's quotient}
  \small
  \begin{block}{Definition}
    Let $\mathbf w$ be a vector, then the Rayleigh's quotient is the scalar ratio :
    $$\mathcal R(\mathbf w)=\frac{\mathbf w^T\mathbf K\mathbf w}{\mathbf w^T\mathbf M\mathbf w}$$
  \end{block}
  \textbf{Properties}
  \begin{enumerate}
    \item Homogeneity. Let $\alpha$ a non-zero constant, then
          $$\mathcal{R}(\alpha\mathbf w)=\frac{(\alpha\mathbf w)^T\mathbf K(\alpha\mathbf w)}{(\alpha\mathbf w)^T\mathbf M(\alpha\mathbf w)}=\frac{\alpha^2\, \mathbf w^T\mathbf K\mathbf w}{\alpha^2\, \mathbf w^T\mathbf M\mathbf w}=\mathcal{R}(\mathbf w)$$
    \item Rayleigh quotient of an eigenvector:
          $$\mathcal{R}(\mathbf w=\mathbf p_i)=\frac{\mathbf p_i^T\mathbf K\mathbf p_i}{\mathbf p_i^T\mathbf M\mathbf p_i}=\lambda_i=\omega_i^2$$
  \end{enumerate}
\end{frame}

\begin{frame}{Stationarity of Rayleigh's quotient}
  \small
  \vspace{-0.2cm}
  \begin{block}{ \small Stationary Rayleigh's quotient in the vicinity of a modal shape}
    \begin{enumerate}
      \setcounter{enumi}{2}
      \item If $\mathbf{w} = \mathbf{p}_i$ then
            \[
              (\mathbf{K} - \mathcal{R}(\mathbf{w}) \mathbf{M})\mathbf{w} = (\mathbf{K} - \lambda_i \mathbf{M}) \mathbf{p}_i  = 0
            \]
    \end{enumerate}
  \end{block}
  \vspace{0.2cm}
  \emph{Sketch of the proof.} First variation of Rayleigh's quotient:
  \begin{align*}
    \delta \mathcal{R}(\mathbf{w})
     & = \frac{2(\bolddelta \mathbf{w}^T\mathbf{K} \mathbf{w})(\mathbf{w}^T\mathbf{M} \mathbf{w})
    - 2(\mathbf{w}^T\mathbf{K} \mathbf{w})(\bolddelta\mathbf{w}^T\mathbf{M} \mathbf{w})}{(\mathbf{w}^T\mathbf{M} \mathbf{w})^2}              \\[0.8em]
     & = \frac{2 \, \bolddelta \mathbf{w}^T}{\mathbf{w}^T\mathbf{M} \mathbf{w}}
    \left( \mathbf{K} \mathbf{w} - \frac{\mathbf{w}^T\mathbf{K} \mathbf{w}}{\mathbf{w}^T\mathbf{M} \mathbf{w}} \mathbf{M} \mathbf{w} \right) \\[0.8em]
     & = \frac{2 \, \bolddelta \mathbf{w}^T}{\mathbf{w}^T\mathbf{M} \mathbf{w}}
    {\color{darkred}(\mathbf{K} - \mathcal{R}(\mathbf{w}) \mathbf{M})\mathbf{w}}
  \end{align*}
  If $\mathbf{w}=\mathbf{p}_i$, then $\mathcal{R}(\mathbf{p}_i)=\lambda_i$ and $(\mathbf{K}-\mathcal{R}(\mathbf{p}_i)\mathbf{M})\mathbf{p}_i
    =(\mathbf{K}-\lambda_i\mathbf{M})\mathbf{p}_i=\mathbf 0$.

  Hence \(\delta\mathcal{R}(\mathbf{p}_i)=0\) for every variation
  \(\bolddelta\mathbf{w}\).
\end{frame}


\begin{frame}{Rayleigh quotient bounding theorem}
  \begin{block}{}
    \begin{enumerate}
      \setcounter{enumi}{3}
      \item Courant-Fischer formulas:
            \[
              \mathcal{R}(\mathbf{w}) \geq \lambda_1 = \min_{\mathbf{w}} \mathcal{R}(\mathbf{w}) \qquad
              \mathcal{R}(\mathbf{w}) \leq \lambda_n = \max_{\mathbf{w}} \mathcal{R}(\mathbf{w})
            \]
      \item Bounding theorem:
            \[
              \lambda_1 \leq \mathcal{R}(\mathbf{w}) \leq \lambda_n
            \]
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Rayleigh principle}
  \emph{Sketch of the proof.}  Modal expansion:
  \[
    \mathbf{w} = z_1 \mathbf{p}_1 + z_2 \mathbf{p}_2 + \cdots + z_i \mathbf{p}_i + \cdots + z_n \mathbf{p}_n = \mathbf{Pz}
  \]
  where
  \begin{align*}
    \mathbf{P} & = [\mathbf{p}_1, \mathbf{p}_2, \dots, \mathbf{p}_i, \dots, \mathbf{p}_n] \quad \text{modal matrix} \\
    \mathbf{z} & = [z_1, z_2, \dots, z_i, \dots, z_n]^T \quad \text{coefficient vector }
  \end{align*}
  Then
  \[
    \mathcal{R}(\mathbf{w} = \mathbf{Pz}) = \frac{\mathbf{z}^T \mathbf{P}^T \mathbf{K} \mathbf{Pz}}{\mathbf{z}^T \mathbf{P}^T \mathbf{M} \mathbf{Pz}} = \frac{\mathbf{z}^T \mathbf{\Lambda} \mathbf{z}}{\mathbf{z}^T \mathbf{z}} =
    \frac{\sum_{i=1}^n \lambda_i z_i^2}{\sum_{i=1}^n z_i^2}
  \]

  \bigskip
  If, without loss of generality, we impose \( \mathbf{w}^T \mathbf{M} \mathbf{w} = 1 \), then $ \mathbf{z}^T \mathbf{z} = \sum_{i=1}^n z_i^2 = 1$ and
  $$ \mathcal{R}(\mathbf{w} = \mathbf{Pz}) = \sum_{i=1}^n \lambda_i z_i^2 $$

\end{frame}

\begin{frame}{Rayleigh principle (continued)}

  \[
    \mathcal{R}(\mathbf{w}) = \sum_{i=1}^n \lambda_i z_i^2 = \lambda_j - \lambda_j + \sum_{i=1}^n \lambda_i z_i^2 = \lambda_j - \lambda_j \left( \sum_{i=1}^n z_i^2 \right) + \sum_{i=1}^n \lambda_i z_i^2
  \]
  \[
    = \lambda_j + \sum_{i=1}^n (\lambda_i - \lambda_j) z_i^2
  \]

  \bigskip

  With eigenvalues ordered as \( \lambda_1 \leq \lambda_2 \leq \cdots \leq \lambda_i \leq \cdots \leq \lambda_n \), we have:
  \begin{itemize}
    \item If \( j = 1 \), then \( \lambda_i - \lambda_1 \geq 0 \) for all \( i \), hence
          \[
            \mathcal{R}(\mathbf{w}) = \lambda_1 + \sum_{i=1}^n (\lambda_i - \lambda_1) z_i^2 \geq \lambda_1.
          \]
    \item If \( j = n \), then \( \lambda_i - \lambda_n \leq 0 \) for all \( i \), hence
          \[
            \mathcal{R}(\mathbf{w}) = \lambda_n + \sum_{i=1}^n (\lambda_i - \lambda_n) z_i^2 \leq \lambda_n.
          \]
  \end{itemize}
\end{frame}



\begin{frame}{Modal frequencies and shapes search via Rayleigh quotient minimization}
  \begin{enumerate}
    \item[(1)] Find $\lambda_1$ by minimization: $$\lambda_1 = \min_{\mathbf{w}}
            \mathcal{R}(\mathbf{w})$$
    \item[(2)] Find $\mathbf p_1$ by solving $(\mathbf{K} - \lambda_1 \mathbf{M}) \mathbf{p}_1  = 0$.
          \vspace{0.3cm}
    \item[(3)] Find $\lambda_2$ by minimization: $$\lambda_2 = \min_{\mathbf{w}} [\mathcal{R}(\mathbf{w}); \, \mathbf{w}^T \mathbf{M} \mathbf{p}_1 = 0]$$
    \item[(4)] Find $\mathbf p_2$ by solving $(\mathbf{K} - \lambda_2 \mathbf{M}) \mathbf{p}_2  = 0$.
          \[
            \vdots
          \]
    \item[(j)]  Find $\lambda_j$ by minimization: $$\lambda_j = \min_{\mathbf{w}} [\mathcal{R}(\mathbf{w}); \, \mathbf{w}^T \mathbf{M} \mathbf{p}_i = 0, \, i = 1, 2, \dots, j-1]$$
    \item[(j+1)] Find $\mathbf p_j$ by solving $(\mathbf{K} - \lambda_j \mathbf{M}) \mathbf{p}_j  = 0$.
  \end{enumerate}
\end{frame}
