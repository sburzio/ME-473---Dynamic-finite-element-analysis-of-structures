\hspace*{\fill}\textbf{\Large 1D models}\hspace*{\fill}

\hrulefill
\begin{itemize}
      \item Displacements: $\mathbf u =\{u_{1}\}$
      \item Strain:  $\boldsymbol\varepsilon =\{\varepsilon_{11}\}$
      \item Stress: $\boldsymbol\sigma =\{\sigma_{11}\}$

      \item Linear strain-displacement relationship: $\varepsilon_{11}=\partial_{x_1}u_{1}$
            $$\nabla=\begin{bmatrix} \partial_{x_1}  \end{bmatrix}$$
      \item Generalised Hooke’s law: $\sigma_{11}=E\varepsilon_{11}$
      \item Stiffness material matrix: $$\mathbf C=\begin{bmatrix} E \end{bmatrix}$$
      \item Dynamic equilibrium equation:
            $$ \partial_{x_1}\sigma_{11} + f_1 = \rho \ddot{u_{1}}\qquad \text{or}\qquad E \partial^2_{x_1x_1}u_{1}+ f_1 = \rho \ddot{u_{1}}$$
\end{itemize}

\hspace*{\fill}\textbf{\Large 2D models - plane stress}\hspace*{\fill}

\hrulefill

\begin{itemize}
      \item Displacements: $\mathbf u =\{u_{1}, u_{2}\}^T$
      \item Strain:  $\boldsymbol\varepsilon =\{\varepsilon_{11}, \varepsilon_{22}, 2\varepsilon_{12}\}^T$, $\varepsilon_{23}=\varepsilon_{13}=0$ and $\varepsilon_{33}=-\frac{\nu}{E}(\sigma_{11}+\sigma_{22})$
      \item Stress: $\boldsymbol\sigma =\{\sigma_{11}, \sigma_{22}, \sigma_{12}\}^T$, $\sigma_{33}=\sigma_{23}=\sigma_{13}=0$
      \item Linear strain-displacement relationship: $\boldsymbol\varepsilon=\nabla\mathbf u$
            $$\nabla=\begin{bmatrix} \partial_{x_1} & 0 \\ 0 & \partial_{x_2} \\ \partial_{x_2} & \partial_{x_1}  \end{bmatrix}$$
      \item Generalised Hooke’s law: $\boldsymbol\sigma=\mathbf C\boldsymbol\varepsilon$
      \item Stiffness isotropic material matrix:
            $$\mathbf C=\frac{E}{1-\nu^2}\begin{bmatrix} 1 & \nu & 0 \\ \nu & 1& 0 \\ 0 & 0 & (1-\nu)/2 \end{bmatrix}$$
      \item Dynamic equilibrium equations:
            $$ \begin{cases}
                        \partial_{x_1}\sigma_{11} + \partial_{x_2}\sigma_{12}  + f_1 = \rho \ddot{u_{1}} \\
                        \partial_{x_2}\sigma_{22} + \partial_{x_1}\sigma_{12}  + f_2 = \rho \ddot{u_{2}}
                  \end{cases}
                  \qquad \text{or}\qquad \nabla^T\mathbf C\nabla\mathbf u + \mathbf f = \rho \ddot{\mathbf u}$$
\end{itemize}

\hspace*{\fill}\textbf{\Large 2D models - plane strain}\hspace*{\fill}

\hrulefill

\begin{itemize}
      \item Displacements: $\mathbf u =\{u_{1}, u_{2}\}^T$, $u_3=0$.
      \item Strain:  $\boldsymbol\varepsilon =\{\varepsilon_{11}, \varepsilon_{22}, 2\varepsilon_{12}\}^T$, $\varepsilon_{33}=\varepsilon_{23}=\varepsilon_{13}=0$
      \item Stress: $\boldsymbol\sigma =\{\sigma_{11}, \sigma_{22}, \sigma_{12}\}^T$, $\sigma_{23}=\sigma_{13}=0$ and $\sigma_{33}=\frac{E\nu(\varepsilon_{11}+\varepsilon_{22})}{(1+\nu)(1+2\nu)}$
      \item Linear strain-displacement relationship: $\boldsymbol\varepsilon=\nabla\mathbf u$
            $$\nabla=\begin{bmatrix} \partial_{x_1} & 0 \\ 0 & \partial_{x_2} \\ \partial_{x_2} & \partial_{x_1}  \end{bmatrix}$$
      \item Generalised Hooke’s law: $\boldsymbol\sigma=\mathbf C\boldsymbol\varepsilon$
      \item Stiffness isotropic material matrix:
            $$\mathbf C=\frac{E}{(1+\nu)(1-2\nu)}\begin{bmatrix} 1-\nu & \nu & 0 \\ \nu & 1-\nu& 0 \\ 0 & 0 & (1-2\nu)/2 \end{bmatrix}$$
      \item Dynamic equilibrium equations:
            $$ \begin{cases}
                        \partial_{x_1}\sigma_{11} + \partial_{x_2}\sigma_{12}  + f_1 = \rho \ddot{u_{1}} \\
                        \partial_{x_2}\sigma_{22} + \partial_{x_1}\sigma_{12}  + f_2 = \rho \ddot{u_{2}}
                  \end{cases}
                  \qquad \text{or}\qquad \nabla^T\mathbf C\nabla\mathbf u + \mathbf f = \rho \ddot{\mathbf u}$$
\end{itemize}

\hspace*{\fill}\textbf{\Large 3D models}\hspace*{\fill}

\hrulefill

\begin{itemize}
      \item Displacements: $\mathbf u =\{u_{1}, u_{2},u_3\}^T$
      \item Strain:  $\boldsymbol\varepsilon =\{\varepsilon_{11}, \varepsilon_{22}, \varepsilon_{33}, 2\varepsilon_{23}, 2\varepsilon_{31}, 2\varepsilon_{12}\}^T$
      \item Stress: $\boldsymbol\sigma =\{\sigma_{11}, \sigma_{22}, \sigma_{33}, 2\sigma_{23}, 2\sigma_{31}, 2\sigma_{12}\}^T$
      \item Linear strain-displacement relationship: $\boldsymbol\varepsilon=\nabla\mathbf u$
            $$\nabla=\begin{bmatrix} \partial_{x_1} & 0 & 0 \\ 0 & \partial_{x_2} & 0 \\ 0 & 0 & \partial_{x_3} \\ 0 & \partial_{x_3} & \partial_{x_2} \\ \partial_{x_3} & 0 & \partial_{x_1}  \\ \partial_{x_2} & \partial_{x_1} & 0 \end{bmatrix}$$
      \item Generalised Hooke’s law: $\boldsymbol\sigma=\mathbf C\boldsymbol\varepsilon$
      \item Stiffness isotropic material matrix:
            $$\mathbf C=\frac{E}{(1+\nu)(1-2\nu)}
                  \begin{bmatrix}
                        1-\nu & \nu   & \nu   & 0          & 0          & 0          \\
                        \nu   & 1-\nu & \nu   & 0          & 0          & 0          \\
                        \nu   & \nu   & 1-\nu & 0          & 0          & 0          \\
                        0     & 0     & 0     & (1-2\nu)/2 & 0          & 0          \\
                        0     & 0     & 0     & 0          & (1-2\nu)/2 & 0          \\
                        0     & 0     & 0     & 0          & 0          & (1-2\nu)/2 \\
                  \end{bmatrix}$$
      \item Dynamic equilibrium equation:
            $$ \begin{cases}
                        \partial_{x_1}\sigma_{11} + \partial_{x_2}\sigma_{12} + \partial_{x_3}\sigma_{13}  + f_1 = \rho \ddot{u_{1}} \\
                        \partial_{x_1}\sigma_{12} +\partial_{x_2}\sigma_{22} + \partial_{x_3}\sigma_{23}  + f_2 = \rho \ddot{u_{2}}  \\
                        \partial_{x_1}\sigma_{13} +\partial_{x_2}\sigma_{23} + \partial_{x_3}\sigma_{33}  + f_3 = \rho \ddot{u_{3}}
                  \end{cases}
                  \qquad \text{or}\qquad \nabla^T\mathbf C\nabla\mathbf u + \mathbf f = \rho \ddot{\mathbf u}$$
\end{itemize}
